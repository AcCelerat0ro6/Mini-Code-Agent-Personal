class Color:
    """ANSI Color codes for terminal output."""
    # Reset
    RESET       = "\033[0m"

    # Regular colors
    BLACK       = "\033[30m"
    RED         = "\033[31m"
    GREEN       = "\033[32m"
    YELLOW      = "\033[33m"
    BLUE        = "\033[34m"
    MAGENTA     = "\033[35m"
    CYAN        = "\033[36m"
    WHITE       = "\033[37m"

    # Bright colors
    BRIGHT_RED     = "\033[91m"
    BRIGHT_GREEN   = "\033[92m"
    BRIGHT_YELLOW  = "\033[93m"
    BRIGHT_BLUE    = "\033[94m"
    BRIGHT_MAGENTA = "\033[95m"
    BRIGHT_CYAN    = "\033[96m"
    BRIGHT_WHITE   = "\033[97m"

    # Background colors
    BG_BLACK    = "\033[40m"
    BG_RED      = "\033[41m"
    BG_GREEN    = "\033[42m"
    BG_YELLOW   = "\033[43m"
    BG_BLUE     = "\033[44m"
    BG_MAGENTA  = "\033[45m"
    BG_CYAN     = "\033[46m"
    BG_WHITE    = "\033[47m"

    # Styles
    BOLD        = "\033[1m"
    DIM         = "\033[2m"
    ITALIC      = "\033[3m"
    UNDERLINE   = "\033[4m"
    BLINK       = "\033[5m"
    REVERSE     = "\033[7m"

    @staticmethod
    def rgb_fg(r: int, g: int, b: int) -> str:
        """Set foreground color using 256 RGB values."""
        return f"\033[38;2;{r};{g};{b}m"

    @staticmethod
    def rgb_bg(r: int, g: int, b: int) -> str:
        """Set background color using 256 RGB values."""
        return f"\033[48;2;{r};{g};{b}m"


def clear_screen():
    """Clear the terminal screen."""
    os.system('cls' if os.name == 'nt' else 'clear')


def colored_text(text: str, color: str) -> str:
    """Wrap text with color codes."""
    return f"{color}{text}{Color.RESET}"


def print_divider(char: str = "═", length: int = 60, color: str = Color.CYAN):
    """Print a decorative divider line."""
    print(colored_text(char * length, color))


def print_centered(text: str, color: str = Color.WHITE, width: int = 60):
    """Print centered colored text."""
    lines = text.split('\n')
    for line in lines:
        padding = max(0, (width - len(line)) // 2)
        print(" " * padding + colored_text(line, color))

def typewriter(text: str, color: str = Color.WHITE,
               speed: float = 0.035, pause_at_end: float = 0.8):
    """
    Display text with a typewriter effect.
    
    Args:
        text: The text to display character by character.
        color: ANSI color code for the text.
        speed: Delay between each character in seconds.
        pause_at_end: Pause after the full text is displayed.
    """
    colored = False
    i = 0
    while i < len(text):
        char = text[i]
        # Handle color escape codes - print them instantly without delay
        if char == '\033':
            # Find the end of the ANSI escape code
            code_end = text.find('m', i)
            if code_end != -1:
                sys.stdout.write(text[i:code_end + 1])
                i = code_end + 1
                continue
        # Handle regular newline
        if char == '\n':
            sys.stdout.write('\n')
            sys.stdout.flush()
            i += 1
            continue
        # Normal character - type with delay
        sys.stdout.write(char)
        sys.stdout.flush()
        time.sleep(speed)
        i += 1
    
    sys.stdout.flush()
    if pause_at_end > 0:
        time.sleep(pause_at_end)


def typewriter_slow(text: str, color: str = Color.WHITE):
    """Slower typewriter for dramatic moments."""
    typewriter(text, color, speed=0.06, pause_at_end=1.2)


def typewriter_fast(text: str, color: str = Color.WHITE):
    """Faster typewriter for urgent/excited dialogue."""
    typewriter(text, color, speed=0.015, pause_at_end=0.5)

def display_portrait(character_id: str, variant: str = None,
                     color: str = Color.WHITE):
    """
    Display an ASCII art portrait for a character.
    
    Args:
        character_id: The character's identifier.
        variant: Optional portrait variant (e.g., 'blush', 'smile').
        color: Color to render the portrait.
    """
    key = character_id if variant is None else f"{character_id}_{variant}"
    portrait = CHARACTER_PORTRAITS.get(key, CHARACTER_PORTRAITS.get(character_id, []))
    
    if not portrait:
        return
    
    # Calculate max line width for frame
    max_width = max(len(line) for line in portrait)
    
    # Print framed portrait
    print(colored_text(f"  ┌{'─' * (max_width + 2)}┐", color))
    for line in portrait:
        padding = max_width - len(line)
        print(colored_text(f"  │ {line}{' ' * padding} │", color))
    print(colored_text(f"  └{'─' * (max_width + 2)}┘", color))

class Character:
    """
    Represents a game character with display info and dialogue capabilities.
    
    Attributes:
        char_id:     Unique identifier for the character.
        name_jp:     Japanese name display.
        name_en:     English name display.
        title:       Character's title/role.
        color:       ANSI color code for name display.
        portrait_id: Key into CHARACTER_PORTRAITS dict.
        affection:   Current affection level with the player.
    """
    
    def __init__(self, char_id: str, name_jp: str, name_en: str,
                 title: str, color: str, portrait_id: str = None):
        self.char_id = char_id
        self.name_jp = name_jp
        self.name_en = name_en
        self.title = title
        self.color = color
        self.portrait_id = portrait_id or char_id
        self.affection = 0  # 0 ~ 100
    
    def speak(self, text: str, variant: str = None,
              speed: float = 0.035, show_portrait: bool = True):
        """
        Character speaks with portrait, name tag, and typewriter effect.
        
        Args:
            text: The dialogue text.
            variant: Optional portrait variant.
            speed: Typewriter speed.
            show_portrait: Whether to display the ASCII portrait.
        """
        if show_portrait:
            display_portrait(self.portrait_id, variant, self.color)
        
        name_display = f"{colored_text(self.name_jp, self.color + Color.BOLD)}"
        subtitle = colored_text(f"「{self.title}」", self.color + Color.DIM)
        
        print(f"  {name_display} {subtitle}")
        print(colored_text("  " + "─" * 50, self.color + Color.DIM))
        
        # Build the full display string with inline color
        dialogue = f"  {self.color}{text}{Color.RESET}"
        typewriter(dialogue, self.color, speed=speed)
        
        if show_portrait:
            print()  # spacing after portrait block
    
    def change_affection(self, amount: int, reason: str = ""):
        """Change affection level, clamped to 0-100."""
        old = self.affection
        self.affection = max(0, min(100, self.affection + amount))
        delta = self.affection - old
        
        if delta > 0:
            indicator = colored_text(f"  ♥ 好感度 +{delta} ({self.name_en})", Color.BRIGHT_RED)
        elif delta < 0:
            indicator = colored_text(f"  ✗ 好感度 {delta} ({self.name_en})", Color.BRIGHT_BLUE)
        else:
            return
        
        if reason:
            indicator += colored_text(f" — {reason}", Color.DIM)
        print(indicator)
        time.sleep(0.6)

class DialogueNode:
    """
    A single dialogue beat within a scene.
    
    Types:
        - "narration"  : Narrator text (no character speaking)
        - "speak"      : A character speaks
        - "choice"     : Player makes a choice
        - "effect"     : Trigger a game effect (affection change, etc.)
        - "clear"      : Clear the screen
        - "portrait"   : Show a character portrait with variant
        - "scene_bg"   : Change scene background description
        - "ending"     : Trigger an ending sequence
    """
    
    def __init__(self, node_type: str, **kwargs):
        self.node_type = node_type
        self.data = kwargs
    
    def __repr__(self):
        return f"DialogueNode({self.node_type}, {self.data})"


class Choice:
    """
    A player choice with text and resulting next node.
    
    Attributes:
        text:      The choice text displayed to the player.
        next_node: The scene node key to jump to if selected.
        effects:   List of (Character, int) affection changes.
    """
    
    def __init__(self, text: str, next_node: str,
                 effects: list = None):
        self.text = text
        self.next_node = next_node
        self.effects = effects or []  # [(character, delta), ...]


class SceneNode:
    """
    A node in the story graph. Contains a sequence of dialogue beats
    and optionally choices that branch to other SceneNodes.
    
    Attributes:
        key:        Unique key for this node.
        dialogue:   List of DialogueNode objects.
        choices:    List of Choice objects (empty = auto-advance).
        next_node:  Key of next node if no choices (auto-advance).
    """
    
    def __init__(self, key: str, dialogue: list = None,
                 choices: list = None, next_node: str = None):
        self.key = key
        self.dialogue = dialogue or []
        self.choices = choices or []
        self.next_node = next_node
