模块结构：

"""

import story_common
import route_yukiko
import route_ami
import route_amamiya

from core import (
    E, scene, narrate, choose, wait,
    yukiko, ami, amamiya,
)

def scene_ending_summary():
    """结局总结"""
    scene("结局总结")
    print("\n" + "=" * 42)
    print("  本次旅程回顾")
    print("=" * 42 + "\n")
    for c in (yukiko, ami, amamiya):
        e = c.ending()
        if e == "true":
            label = "★ True"
        elif e == "good":
            label = "♥ Good"
        else:
            label = "○ Normal"
        print(f"  {c.name}: {label}  (好感度 {c.affection}/100)")
    print()
    narrate("感谢游玩本演示！")
    print("  提示：参与「星降传说」的许愿且好感度 ≥ 85 可解锁 True 结局。")
    print("  重新运行 python game.py，可以尝试不同的选择，攻略其他角色！")
    print("=" * 42 + "\n")
    wait("按 Enter 退出…")


def main():
    """主循环：通过场景字典与返回值驱动剧情流程"""
    current = "intro"
    visited = set()

    try:
        while current:
            if current in visited:
                print(f"[警告] 场景 '{current}' 重复触发，已终止以避免死循环。")
                break
            visited.add(current)

            if current not in SCENES:
                print(f"[错误] 未找到场景: {current}")
                break

            current = SCENES[current]()
    except KeyboardInterrupt:
        E.clear_screen()
        print("\n感谢游玩，下次再见！\n")


if __name__ == "__main__":
    main()
